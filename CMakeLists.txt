cmake_minimum_required(VERSION 3.20)
project(AsterCore VERSION 0.1.0 LANGUAGES CXX)
option(ASTER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(aster_core aster/core/search.cc aster/core/workspace.cc)
target_include_directories(aster_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(aster_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(aster_core PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(aster_core PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
  if(ASTER_SANITIZERS)
    target_compile_options(aster_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(aster_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(aster_core_tests aster/tests/core_test.cc)
target_link_libraries(aster_core_tests PRIVATE aster_core)
if(MSVC)
  target_compile_options(aster_core_tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(aster_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME aster_core COMMAND aster_core_tests)
